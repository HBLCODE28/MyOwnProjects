MovieMeet — הקמת מסד נתונים
============================

1. בפאנל האחסון צרי מסד נתונים ומשתמש MySQL, ותני למשתמש הרשאות מלאות למסד.
2. החליפי את `moviemeet` בשם מסד הנתונים שקיבלת מהאחסון, אם הוא שונה.
3. בחרי את מסד הנתונים ב-phpMyAdmin, פתחי SQL והדביקי את כל הקוד הבא.
4. עדכני את אותם פרטים גם בקובץ `htdocs/api/config.php` בשרת.

-- הריצי את שתי השורות הבאות רק אם לפאנל שלך מותר ליצור מסד נתונים מתוך phpMyAdmin.
CREATE DATABASE IF NOT EXISTS `moviemeet` CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE `moviemeet`;

CREATE TABLE IF NOT EXISTS mm_users (
  id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  name VARCHAR(80) NOT NULL,
  email VARCHAR(254) NOT NULL UNIQUE,
  password_hash VARCHAR(255) NOT NULL,
  created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE IF NOT EXISTS mm_sessions (
  token CHAR(64) PRIMARY KEY,
  user_id INT UNSIGNED NOT NULL,
  expires_at DATETIME NOT NULL,
  created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
  INDEX(user_id),
  CONSTRAINT mm_sessions_user FOREIGN KEY(user_id) REFERENCES mm_users(id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE IF NOT EXISTS mm_events (
  id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  token CHAR(32) NOT NULL UNIQUE,
  organizer_id INT UNSIGNED NOT NULL,
  name VARCHAR(120) NOT NULL,
  date_start DATE NOT NULL,
  date_end DATE NOT NULL,
  created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT mm_events_user FOREIGN KEY(organizer_id) REFERENCES mm_users(id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE IF NOT EXISTS mm_participants (
  id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  event_id INT UNSIGNED NOT NULL,
  user_id INT UNSIGNED NULL,
  name VARCHAR(80) NOT NULL,
  city VARCHAR(80) NOT NULL,
  availability LONGTEXT NOT NULL,
  genres LONGTEXT NOT NULL,
  created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
  UNIQUE KEY mm_participant_user (event_id,user_id),
  CONSTRAINT mm_participants_event FOREIGN KEY(event_id) REFERENCES mm_events(id) ON DELETE CASCADE,
  CONSTRAINT mm_participants_user FOREIGN KEY(user_id) REFERENCES mm_users(id) ON DELETE SET NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE IF NOT EXISTS mm_cinemas (
  id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  network VARCHAR(50) NOT NULL,
  name VARCHAR(100) NOT NULL,
  city VARCHAR(80) NOT NULL,
  latitude DECIMAL(10,7) NOT NULL,
  longitude DECIMAL(10,7) NOT NULL,
  booking_url VARCHAR(500) NOT NULL,
  UNIQUE KEY mm_cinema_name (network,name)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE IF NOT EXISTS mm_candidates (
  id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  event_id INT UNSIGNED NOT NULL,
  cinema_id INT UNSIGNED NOT NULL,
  tmdb_id INT NULL,
  movie_title VARCHAR(180) NOT NULL,
  poster_path VARCHAR(300) NULL,
  genres LONGTEXT NOT NULL,
  show_date DATE NOT NULL,
  show_time TIME NOT NULL,
  booking_url VARCHAR(500) NOT NULL,
  seats_checked TINYINT(1) NOT NULL DEFAULT 0,
  created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT mm_candidates_event FOREIGN KEY(event_id) REFERENCES mm_events(id) ON DELETE CASCADE,
  CONSTRAINT mm_candidates_cinema FOREIGN KEY(cinema_id) REFERENCES mm_cinemas(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE IF NOT EXISTS mm_votes (
  participant_id INT UNSIGNED NOT NULL,
  candidate_id INT UNSIGNED NOT NULL,
  vote TINYINT NOT NULL,
  created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY(participant_id,candidate_id),
  CONSTRAINT mm_votes_participant FOREIGN KEY(participant_id) REFERENCES mm_participants(id) ON DELETE CASCADE,
  CONSTRAINT mm_votes_candidate FOREIGN KEY(candidate_id) REFERENCES mm_candidates(id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
